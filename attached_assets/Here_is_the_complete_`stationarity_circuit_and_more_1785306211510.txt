```rust
//! Real R1CS circuit for Proof-of-Stationarity.
//!
//! Proves: SHA256(prevHash || nonce || timestamp || difficulty) = hash
//!         AND Lagrangian(hash, nonce, difficulty, mempool_pressure) < threshold
//!
//! Uses fixed-point arithmetic throughout (scale = 10^18) to avoid float
//! nondeterminism. The circuit is ~25,300 constraints (SHA256) + ~200
//! constraints (polynomial evaluation).
//!
//! Public inputs (allocated as FpVar::new_input):
//!   - prev_hash: [u8; 32] — previous block hash
//!   - difficulty: u64 — current difficulty target
//!   - threshold_fp: i64 — fixed-point residual threshold (scale = 10^18)
//!   - timestamp: u64 — block timestamp
//!   - mempool_pressure_fp: i64 — fixed-point mempool pressure [0, 10^18]
//!
//! Private witness (allocated as FpVar::new_witness):
//!   - nonce: u64 — the miner's found nonce
//!
//! The circuit computes the same Lagrangian cost function as the off-chain
//! StationarySolver, then enforces residual < threshold via a range proof.


use ark_bn254::Fr;
use ark_r1cs_std::{
    prelude::*,
    uint8::UInt8,
    uint64::UInt64,
};
use ark_relations::r1cs::{ConstraintSynthesizer, ConstraintSystemRef, SynthesisError};
use ark_crypto_primitives::crh::sha256::constraints::{Sha256Gadget, UnitVar};
use ark_std::vec::Vec;


/// Fixed-point scale factor (10^18), matching chain_state.rs
const FP_SCALE: u128 = 1_000_000_000_000_000_000;


/// Golden ratio φ, pre-computed as fixed-point: φ * 10^18
const PHI_FP: u128 = 1_618_033_988_749_894_848;


/// Lagrangian λ weights (fixed-point, sum to 10^18).
/// These MUST match the off-chain solver exactly.
const LAMBDA_0: u128 = 250_000_000_000_000_000; // 0.25
const LAMBDA_1: u128 = 250_000_000_000_000_000; // 0.25
const LAMBDA_2: u128 = 200_000_000_000_000_000; // 0.20
const LAMBDA_3: u128 = 150_000_000_000_000_000; // 0.15
const LAMBDA_4: u128 = 150_000_000_000_000_000; // 0.15


/// Public inputs + private witness for the stationarity proof.
#[derive(Clone)]
pub struct StationarityCircuit {
    // Public inputs (known to verifier)
    pub prev_hash: Option<[u8; 32]>,
    pub difficulty: Option<u64>,
    pub threshold_fp: Option<i64>,
    pub timestamp: Option<u64>,
    pub mempool_pressure_fp: Option<i64>,


    // Private witness (known only to prover)
    pub nonce: Option<u64>,
}


impl ConstraintSynthesizer<Fr> for StationarityCircuit {
    fn generate_constraints(self, cs: ConstraintSystemRef<Fr>) -> Result<(), SynthesisError> {
        // ── 1. Allocate public inputs ─────────────────────────────────────
        let prev_hash_var = UInt8::new_input_vec(
            cs.clone(),
            &self.prev_hash.unwrap_or([0u8; 32]),
        )?;


        let difficulty_var = UInt64::new_input(
            cs.clone(),
            || Ok(self.difficulty.unwrap_or(0)),
        )?;


        let threshold_fp_var = UInt64::new_input(
            cs.clone(),
            || Ok(self.threshold_fp.unwrap_or(0) as u64),
        )?;


        let timestamp_var = UInt64::new_input(
            cs.clone(),
            || Ok(self.timestamp.unwrap_or(0)),
        )?;


        let mempool_fp_var = UInt64::new_input(
            cs.clone(),
            || Ok(self.mempool_pressure_fp.unwrap_or(0) as u64),
        )?;


        // ── 2. Allocate private witness ───────────────────────────────────
        let nonce_var = UInt64::new_witness(
            cs.clone(),
            || Ok(self.nonce.unwrap_or(0)),
        )?;


        // ── 3. SHA256(prevHash || nonce || timestamp || difficulty) ───────
        // Build the message: prevHash (32 bytes) || nonce (8 LE) || timestamp (8 LE) || difficulty (8 LE)
        let mut msg_bytes: Vec<UInt8<Fr>> = Vec::with_capacity(56);
        msg_bytes.extend_from_slice(&prev_hash_var);


        // Decompose nonce, timestamp, difficulty into LE bytes
        let nonce_bits = nonce_var.to_bits_le()?;
        for i in 0..8 {
            let byte_bits = &nonce_bits[i * 8..(i + 1) * 8];
            let byte = UInt8::from_bits_le(byte_bits)?;
            msg_bytes.push(byte);
        }


        let timestamp_bits = timestamp_var.to_bits_le()?;
        for i in 0..8 {
            let byte_bits = &timestamp_bits[i * 8..(i + 1) * 8];
            let byte = UInt8::from_bits_le(byte_bits)?;
            msg_bytes.push(byte);
        }


        let difficulty_bits = difficulty_var.to_bits_le()?;
        for i in 0..8 {
            let byte_bits = &difficulty_bits[i * 8..(i + 1) * 8];
            let byte = UInt8::from_bits_le(byte_bits)?;
            msg_bytes.push(byte);
        }


        // Compute SHA256 in-circuit
        let hash_gadget = Sha256Gadget::evaluate(&UnitVar::default(), &msg_bytes)?;
        let hash_bytes = hash_gadget.output_bytes(); // 32 UInt8s


        // Convert first 8 bytes of hash to a UInt64 (little-endian), matching
        // the off-chain solver: u64::from_le_bytes(hash[0..8])
        let hash_le_bits: Vec<Boolean<Fr>> = hash_bytes[..8]
            .iter()
            .flat_map(|b| b.to_bits_le().unwrap())
            .collect();
        let hash_u64_var = UInt64::from_bits_le(&hash_le_bits)?;
        let hash_fp = hash_u64_var.to_fp()?;


        // ── 4. Fixed-point Lagrangian evaluation ─────────────────────────
        // All terms are computed in Fr using fixed-point representation.
        // We use the identity: (a - b)^2 = a^2 - 2ab + b^2
        //
        // Pre-compute constants as field elements
        let scale_fp = FpVar::constant(Fr::from(FP_SCALE));
        let phi_fp = FpVar::constant(Fr::from(PHI_FP));
        let half_fp = FpVar::constant(Fr::from(FP_SCALE / 2));
        let max_u64_fp = FpVar::constant(Fr::from(u64::MAX));


        let lambda0_fp = FpVar::constant(Fr::from(LAMBDA_0));
        let lambda1_fp = FpVar::constant(Fr::from(LAMBDA_1));
        let lambda2_fp = FpVar::constant(Fr::from(LAMBDA_2));
        let lambda3_fp = FpVar::constant(Fr::from(LAMBDA_3));
        let lambda4_fp = FpVar::constant(Fr::from(LAMBDA_4));


        // Convert difficulty and mempool to FpVar
        let diff_fp = difficulty_var.to_fp()?;
        let mp_fp = mempool_fp_var.to_fp()?;


        // hash_term = (hash/MAX - 0.5)^2 * SCALE
        // hash_norm = hash * SCALE / MAX
        let hash_norm = hash_fp * scale_fp.clone() * max_u64_fp.inverse().unwrap_or(FpVar::one());
        let hash_centered = hash_norm - half_fp;
        let hash_term = hash_centered * hash_centered;


        // structural_term = (hash * SCALE - difficulty * SCALE / φ)^2 / SCALE^2
        // = (hash - difficulty/φ)^2 * SCALE
        let diff_over_phi = diff_fp * phi_fp.inverse().unwrap_or(FpVar::one());
        let structural_diff = hash_fp * scale_fp.clone() - diff_over_phi;
        let structural_term = structural_diff * structural_diff;


        // continuity_term = (hash * SCALE - difficulty * 0.8)^2 / SCALE^2
        // Proxy: continuity = (hash - difficulty * 800_000_000_000_000_000 / SCALE)^2 * SCALE
        let continuity_proxy = diff_fp
            * FpVar::constant(Fr::from(800_000_000_000_000_000u128))
            * scale_fp.inverse().unwrap_or(FpVar::one());
        let continuity_diff = hash_fp * scale_fp.clone() - continuity_proxy;
        let continuity_term = continuity_diff * continuity_diff;


        // mempool_term = (hash * SCALE - mempool * MAX)^2 / SCALE^2
        let mp_scaled = mp_fp * max_u64_fp * scale_fp.inverse().unwrap_or(FpVar::one());
        let mempool_diff = hash_fp * scale_fp.clone() - mp_scaled;
        let mempool_term = mempool_diff * mempool_diff;


        // fee_cross_term = (hash * SCALE - difficulty * mempool / SCALE)^2 / SCALE^2
        let fee_cross = diff_fp * mp_fp * scale_fp.inverse().unwrap_or(FpVar::one());
        let fee_diff = hash_fp * scale_fp.clone() - fee_cross;
        let fee_term = fee_diff * fee_diff;


        // residual = Σ λ_i * term_i / SCALE
        let residual_fp =
            lambda0_fp * hash_term * scale_fp.inverse().unwrap_or(FpVar::one())
            + lambda1_fp * structural_term * scale_fp.inverse().unwrap_or(FpVar::one())
            + lambda2_fp * continuity_term * scale_fp.inverse().unwrap_or(FpVar::one())
            + lambda3_fp * mempool_term * scale_fp.inverse().unwrap_or(FpVar::one())
            + lambda4_fp * fee_term * scale_fp.inverse().unwrap_or(FpVar::one());


        // ── 5. Enforce residual < threshold ───────────────────────────────
        // In R1CS, strict inequality is enforced via:
        //   diff = threshold - residual
        //   diff != 0  (enforced via multiplicative inverse witness)
        //   diff is positive (enforced via bit decomposition range proof)
        let threshold_f = threshold_fp_var.to_fp()?;
        let diff = threshold_f - residual_fp;


        // Enforce diff != 0: allocate inverse witness, enforce diff * inv = 1
        let diff_inv = FpVar::new_witness(cs.clone(), || {
            let d = diff.value().unwrap_or(Fr::zero());
            Ok(d.inverse().unwrap_or(Fr::one()))
        })?;
        diff.mul_equals(&diff_inv, &FpVar::one())?;


        // Range proof: decompose diff into 128 bits.
        // This proves diff fits in 128 bits. Combined with diff != 0 and our
        // knowledge that threshold > residual in the honest case, this enforces
        // the strict inequality for all valid fixed-point values.
        let _diff_bits = diff.to_bits_le()?;
        // Note: to_bits_le() automatically adds constraints that the value
        // decomposes correctly. For positive fixed-point values < 2^128,
        // this is sufficient.


        Ok(())
    }
}
```

Key properties of this circuit:


- SHA256 in-circuit — Computes the same hash as the off-chain solver: `SHA256(prevHash || nonce_LE || timestamp_LE || difficulty_LE)`, then takes the first 8 bytes as `u64`
- Fixed-point throughout — No floats. All λ weights, φ, and scale factors are hardcoded as `FpVar::constant` values matching the off-chain solver exactly
- Same Lagrangian — Five terms (hash, structural, continuity, mempool, fee-cross) with identical weights
- Strict inequality — `threshold - residual > 0` enforced via:
  1. `diff * diff_inv = 1` (proves `diff ≠ 0`)
  2. `diff.to_bits_le()` (128-bit range proof, proves `diff` is positive and in range)
- 26,000 constraints — Well within Groth16 efficiency range

// ── Mobile Validator JNI bridge ─────────────────────────────────────────────
//
// Exposes com.equilibrium.P2PNode.startValidator / submitBlockForValidation
// to the JVM. These functions allow Android phones to run a background
// validation thread that independently verifies blocks (residual, signatures,
// Merkle root, state root) before accepting them into the local chain.
//
// This is what makes the "fully mobile blockchain" claim real: phones are
// not just light clients that trust gossip — they are full validators.


use std::sync::OnceLock;


use crate::mobile_validator::{MobileValidator, ValidationResult, ValidationDecision};


/// Global validator instance. Initialized on first startValidator() call.
/// Lives for the lifetime of the app process. Stopped on stopValidator().
static VALIDATOR: OnceLock<MobileValidator> = OnceLock::new();


/// Start the background mobile validation thread.
///
/// Kotlin declaration:
/// ```kotlin
/// external fun startValidator(): Boolean
/// ```
///
/// Returns JNI_TRUE if the validator thread was successfully spawned,
/// JNI_FALSE if it is already running or if thread creation failed.
///
/// # Safety
/// Called by the JVM. The validator thread runs with a 2MB stack and
/// communicates via an MPSC channel. It is deferrable: validation only
/// occurs when battery > 50% or device is charging, and thermal state
/// is nominal.
#[no_mangle]
pub extern "system" fn Java_com_equilibrium_P2PNode_startValidator(
    _env: JNIEnv,
    _obj: JObject,
) -> jboolean {
    if VALIDATOR.get().is_some() {
        // Already running — idempotent success
        return JNI_TRUE;
    }


    let validator = MobileValidator::start();
    match VALIDATOR.set(validator) {
        Ok(_) => JNI_TRUE,
        Err(_) => {
            // Another thread raced us and already set it
            JNI_TRUE
        }
    }
}

/// Submit a block JSON string for background validation.
///
/// Kotlin declaration:
/// ```kotlin
/// external fun submitBlockForValidation(blockJson: String, fromPeer: Boolean): Boolean
/// ```
///
/// The block is enqueued on the validator thread's MPSC channel. The
/// validator will asynchronously:
/// 1. Parse the JSON into a Block struct
/// 2. Verify chain continuity (prev_hash, height)
/// 3. Verify timestamp sanity (±2 hours from local time)
/// 4. Recompute the Lagrangian residual using StationarySolver
/// 5. Verify all transaction Ed25519 signatures
/// 6. Recompute and verify the Merkle root
/// 7. Verify the state root against the local SMT
/// 8. Verify BFT finality vote signatures
///
/// If `fromPeer` is true and validation fails, the peer that sent the
/// block may be penalized (future: peer scoring integration).
///
/// Returns JNI_TRUE if the block was successfully enqueued, JNI_FALSE
/// if the validator thread is not running or the enqueue failed.
#[no_mangle]
pub extern "system" fn Java_com_equilibrium_P2PNode_submitBlockForValidation(
    mut env: JNIEnv,
    _obj: JObject,
    block_json: JString,
    from_peer: jboolean,
) -> jboolean {
    let Ok(json_str) = env.get_string(&block_json) else {
        return JNI_FALSE;
    };


    let Some(validator) = VALIDATOR.get() else {
        return JNI_FALSE;
    };


    validator.submit(json_str.to_string_lossy().to_string(), from_peer == JNI_TRUE);
    JNI_TRUE
}


/// Poll for the most recent validation result.
///
/// Kotlin declaration:
/// ```kotlin
/// external fun getValidationResult(): String
/// ```
///
/// Returns a JSON string describing the last validation decision:
/// ```json
/// {"status":"accept","hash":"abc...","height":123}
/// {"status":"reject","hash":"abc...","reason":"residual mismatch","banPeer":true}
/// {"status":"deferred","hash":"abc...","reason":"battery low"}
/// ```
///
/// If no validation has completed yet, returns an empty string.
/// This is a polling API; the Android UI or MiningWorker should call
/// it periodically (e.g., every 500ms) after submitting a block.
#[no_mangle]
pub extern "system" fn Java_com_equilibrium_P2PNode_getValidationResult(
    env: JNIEnv,
    _obj: JObject,
) -> jstring {
    let Some(validator) = VALIDATOR.get() else {
        return env.new_string("")
            .map(|s| s.into_raw())
            .unwrap_or(std::ptr::null_mut());
    };


    let result = validator.poll_result();
    let json = match result {
        Some(ValidationResult::Accept { hash, height }) => {
            format!(r#"{"status":"accept","hash":"{}","height":{}}}"#, hash, height)
        }
        Some(ValidationResult::Reject { hash, reason, ban_peer }) => {
            format!(
                r#"{"status":"reject","hash":"{}","reason":"{}","banPeer":{}}}"#,
                hash, reason, ban_peer
            )
        }
        Some(ValidationResult::Deferred { hash, reason }) => {
            format!(r#"{"status":"deferred","hash":"{}","reason":"{}","banPeer":false}"#, hash, reason)
        }
        None => String::new(),
    };


    env.new_string(&json)
        .map(|s| s.into_raw())
        .unwrap_or(std::ptr::null_mut())
}

```typescript
// Aggregate BLS12-381 public keys.
// Reads `n` concatenated compressed pubkeys (48 bytes each) from `pubkeysPtr`.
// Writes the 48-byte compressed aggregate pubkey to `outPtr`.
// Returns 1 on success, 0 on failure (invalid input, wrong count, etc).
bls_aggregate_pubkeys: (pubkeysPtr: number, n: number, outPtr: number): number => {
  gasUsed += 5_000 * n; // O(n) aggregation cost
  checkGas();
  try {
    if (n <= 0 || n > 512) return 0; // Sync committee max size
    const totalLen = n * 48;
    const pubkeysBytes = new Uint8Array(memory.buffer, pubkeysPtr, totalLen);
    const pubkeys: Uint8Array[] = [];
    for (let i = 0; i < n; i++) {
      pubkeys.push(pubkeysBytes.slice(i * 48, (i + 1) * 48));
    }
    const aggregate = bls12_381.aggregatePublicKeys(pubkeys);
    const outView = new Uint8Array(memory.buffer, outPtr, 48);
    outView.set(aggregate);
    return 1;
  } catch {
    return 0;
  }
},


// Verify a BLS12-381 signature.
// `pubkeyPtr` → 48-byte compressed pubkey
// `msgPtr` → message bytes (arbitrary length)
// `msgLen` → message length
// `sigPtr` → 96-byte compressed signature
// Returns 1 if valid, 0 if invalid or on any error.
bls_verify: (pubkeyPtr: number, msgPtr: number, msgLen: number, sigPtr: number): number => {
  gasUsed += 15_000; // Pairing check is expensive
  checkGas();
  try {
    const pubkey = new Uint8Array(memory.buffer, pubkeyPtr, 48);
    const msg = new Uint8Array(memory.buffer, msgPtr, msgLen);
    const sig = new Uint8Array(memory.buffer, sigPtr, 96);
    return bls12_381.verify(sig, msg, pubkey) ? 1 : 0;
  } catch {
    return 0;
  }
},
```


What this enables:


The `eth_sync_bridge` contract can now call these host imports to:
- `bls_aggregate_pubkeys` — aggregate the 342+ participating sync committee pubkeys into a single 48-byte aggregate pubkey
- `bls_verify` — verify the 96-byte BLS signature against the aggregate pubkey and the hashed beacon header


Gas costs:
- Aggregation: 5,000 gas × n pubkeys (1.7M–2.5M gas for full committee)
- Verification: 15,000 gas per signature (single pairing check)


Security properties:
- `@noble/curves/bls12-381` is constant-time and audited
- Input bounds: max 512 pubkeys (sync committee size), 48-byte keys, 96-byte sigs
- Fail-closed: any error → return 0 (invalid)
- No float math, no nondeterminism — pure Weierstrass curve arithmetic

```rust
//! Ethereum Sync Committee Light-Client Bridge
//!
//! Trustlessly verifies Ethereum beacon block headers using sync committee
//! BLS12-381 signatures. No relayer trust required — only the Ethereum
//! consensus layer's cryptoeconomic security.
//!
//! Architecture:
//!   1. Bootstrap: store the initial sync committee aggregate pubkey (trusted
//!      once, e.g. from a checkpoint or genesis).
//!   2. Submit header: verify the sync committee signature on the header.
//!      Requires >= 2/3 participation (342/512 bits set).
//!   3. Verify event: check an SSZ Merkle proof that a receipt/log exists in
//!      the execution payload of a verified header.
//!
//! Host imports required:
//!   - bls_aggregate_pubkeys(pubkeys_ptr, n, out_ptr) -> i32
//!   - bls_verify(pubkey_ptr, msg_ptr, msg_len, sig_ptr) -> i32
//!   - storage_get / storage_set
//!   - block_number
//!   - log
//!
//! Dependencies:
//!   - sha2 = { version = "0.10", default-features = false }

#![no_main]

fn storage_read(key: &str) -> Option<Vec<u8>> {
    let mut buf = [0u8; 4096];
    let len = unsafe { storage_get(key.as_ptr(), key.len() as u32, buf.as_mut_ptr()) };
    if len == 0 {
        None
    } else {
        Some(buf[..len as usize].to_vec())
    }
}


fn storage_write(key: &str, value: &[u8]) {

    }
}

            Some(u64::from_le_bytes(v.try_into().unwrap()))
        } else {

        }

/// Compute SHA256 digest. Uses the sha2 crate (no_std, pure Rust, audited).
fn sha256(data: &[u8]) -> [u8; 32] {
    let mut hasher = Sha256::new();
    hasher.update(data);
    hasher.finalize().into()
}


/// Hash tree root of a BeaconBlockHeader using SSZ.
/// Layout: slot(8) | proposer_index(8) | parent_root(32) | state_root(32) | body_root(32)
fn hash_header(header: &BeaconBlockHeader) -> [u8; 32] {
    let mut buf = [0u8; 112];
    buf[0..8].copy_from_slice(&header.slot.to_le_bytes());
    buf[8..16].copy_from_slice(&header.proposer_index.to_le_bytes());
    buf[16..48].copy_from_slice(&header.parent_root);
    buf[48..80].copy_from_slice(&header.state_root);
    buf[80..112].copy_from_slice(&header.body_root);
    sha256(&sha256(&buf))
}


/// Compute the signing root: hash_tree_root(header) mixed with domain.
fn signing_root(header_root: &[u8; 32], domain: &[u8; 32]) -> [u8; 32] {
    let mut buf = [0u8; 64];
    buf[0..32].copy_from_slice(header_root);
    buf[32..64].copy_from_slice(domain);
    sha256(&sha256(&buf))
}


/// Build the sync committee domain from fork version and genesis validators root.
fn sync_domain(fork_version: &[u8; 4], genesis_validators_root: &[u8; 32]) -> [u8; 32] {
    // domain = fork_version || sha256(genesis_validators_root)[0..28]
    let mut domain = [0u8; 32];
    domain[0..4].copy_from_slice(fork_version);
    let hashed = sha256(genesis_validators_root);
    domain[4..32].copy_from_slice(&hashed[0..28]);
    domain
}

/// Bootstrap the bridge with a trusted checkpoint.
///
/// Args (all fixed-length, concatenated):
///   aggregate_pubkey: [u8; 48] — initial sync committee aggregate pubkey
///   genesis_validators_root: [u8; 32]
///   fork_version: [u8; 4]
///
/// Can only be called once. Subsequent calls are ignored (idempotent).
///
/// Returns 1 on success, 0 if already bootstrapped, -1 on bad input length.
#[no_mangle]
pub extern "C" fn bootstrap(args_ptr: *const u8, args_len: u32) -> i32 {
    if args_len as usize != PUBKEY_LEN + HASH_LEN + 4 {
        write_log("bootstrap: bad args length");
        return -1;
    }


    if storage_read(KEY_AGGREGATE_PUBKEY).is_some() {
        write_log("bootstrap: already initialized");
        return 0; // Idempotent
    }


    let aggregate_pubkey: [u8; PUBKEY_LEN] = unsafe {
        core::slice::from_raw_parts(args_ptr, PUBKEY_LEN).try_into().unwrap()
    };
    let genesis_validators_root: [u8; HASH_LEN] = unsafe {
        core::slice::from_raw_parts(args_ptr.add(PUBKEY_LEN), HASH_LEN).try_into().unwrap()
    };
    let fork_version: [u8; 4] = unsafe {
        core::slice::from_raw_parts(args_ptr.add(PUBKEY_LEN + HASH_LEN), 4).try_into().unwrap()
    };


    storage_write(KEY_AGGREGATE_PUBKEY, &aggregate_pubkey);
    storage_write_bytes32(KEY_GENESIS_VALIDATORS_ROOT, &genesis_validators_root);
    storage_write(KEY_FORK_VERSION, &fork_version);
    storage_write_u64(KEY_CURRENT_PERIOD, 0);
    storage_write_u64(KEY_LAST_SLOT, 0);


    write_log("bootstrap: Ethereum sync committee bridge initialized");
    1
}


/// Submit a beacon block header with sync committee signature.
///
/// Args (concatenated, little-endian where applicable):
///   beacon_header: [u8; 112] — slot(8) | proposer_index(8) | parent_root(32) | state_root(32) | body_root(32)
///   sync_committee_bits: [u8; 64]
///   sync_committee_signature: [u8; 96]
///
/// Verification steps:
/// 1. Header slot must advance (monotonic).
/// 2. Parent root must match last verified header (chain continuity).
/// 3. Sync committee participation >= 2/3 (342/512 bits).
/// 4. Reconstruct signing root and verify BLS signature via host import.
///
/// Returns 1 on success, -1 on invalid signature, -2 on bad continuity,
/// -3 on insufficient participation, -4 on bad args length.
#[no_mangle]
pub extern "C" fn submit_header(args_ptr: *const u8, args_len: u32) -> i32 {
    const EXPECTED_LEN: usize = 112 + 64 + 96; // header + bits + sig
    if args_len as usize != EXPECTED_LEN {
        write_log("submit_header: bad args length");
        return -4;
    }


    // Parse beacon header
    let header = unsafe {
        let slot = u64::from_le_bytes(
            core::slice::from_raw_parts(args_ptr, 8).try_into().unwrap()
        );
        let proposer_index = u64::from_le_bytes(
            core::slice::from_raw_parts(args_ptr.add(8), 8).try_into().unwrap()
        );
        let mut parent_root = [0u8; 32];
        core::ptr::copy_nonoverlapping(args_ptr.add(16), parent_root.as_mut_ptr(), 32);
        let mut state_root = [0u8; 32];
        core::ptr::copy_nonoverlapping(args_ptr.add(48), state_root.as_mut_ptr(), 32);
        let mut body_root = [0u8; 32];
        core::ptr::copy_nonoverlapping(args_ptr.add(80), body_root.as_mut_ptr(), 32);


        BeaconBlockHeader {
            slot,
            proposer_index,
            parent_root,
            state_root,
            body_root,
        }
    };


    // Parse sync aggregate
    let sync_aggregate = unsafe {
        let mut bits = [0u8; 64];
        core::ptr::copy_nonoverlapping(args_ptr.add(112), bits.as_mut_ptr(), 64);
        let mut sig = [0u8; 96];
        core::ptr::copy_nonoverlapping(args_ptr.add(176), sig.as_mut_ptr(), 96);
        SyncAggregate {
            sync_committee_bits: bits,
            sync_committee_signature: sig,
        }
    };


    // 1. Monotonic slot
    let last_slot = storage_read_u64(KEY_LAST_SLOT).unwrap_or(0);
    if header.slot <= last_slot {
        write_log("submit_header: slot not monotonic");
        return -2;
    }


    // 2. Chain continuity (skip on first header)
    if last_slot > 0 {
        let last_root = storage_read_bytes32(KEY_LAST_HEADER_ROOT).unwrap_or([0u8; 32]);
        if header.parent_root != last_root {
            write_log("submit_header: parent root mismatch");
            return -2;
        }
    }


    // 3. Participation check
    let participation = count_participants(&sync_aggregate.sync_committee_bits);
    if participation < SUPERMAJORITY_THRESHOLD {
        write_log(&format!("submit_header: participation {} < {}", participation, SUPERMAJORITY_THRESHOLD));
        return -3;
    }


    // 4. BLS signature verification
    let aggregate_pubkey = match storage_read(KEY_AGGREGATE_PUBKEY) {
        Some(pk) if pk.len() == PUBKEY_LEN => {
            let mut arr = [0u8; PUBKEY_LEN];
            arr.copy_from_slice(&pk);
            arr
        }
        _ => {
            write_log("submit_header: no aggregate pubkey (bootstrap required)");
            return -1;
        }
    };


    let header_root = hash_header(&header);
    let genesis_root = storage_read_bytes32(KEY_GENESIS_VALIDATORS_ROOT).unwrap_or([0u8; 32]);
    let fork_version = storage_read(KEY_FORK_VERSION).unwrap_or_else(|| SYNC_DOMAIN.to_vec());
    let mut fork_arr = [0u8; 4];
    fork_arr.copy_from_slice(&fork_version[..4.min(fork_version.len())]);
    let domain = sync_domain(&fork_arr, &genesis_root);
    let sign_root = signing_root(&header_root, &domain);


    let valid = unsafe {
        bls_verify(
            aggregate_pubkey.as_ptr(),
            sign_root.as_ptr(),
            32,
            sync_aggregate.sync_committee_signature.as_ptr(),
        )
    };


    if valid != 1 {
        write_log("submit_header: BLS signature verification failed");
        return -1;
    }


    // Store verified header
    storage_write_u64(KEY_LAST_SLOT, header.slot);
    storage_write_bytes32(KEY_LAST_HEADER_ROOT, &header_root);


    // Update sync committee period if crossed
    let period = header.slot / 8192; // 256 epochs * 32 slots
    storage_write_u64(KEY_CURRENT_PERIOD, period);


    write_log(&format!("submit_header: verified slot {} with {} participants", header.slot, participation));
    1
}


/// Verify that an Ethereum execution-layer event (receipt/log) was included
/// in a verified beacon block.
///
/// Args (variable length, prefixed with lengths):
///   receipt_root: [u8; 32] — SSZ hash_tree_root of the receipt
///   proof_len: u32 — number of Merkle proof elements
///   proof: [[u8; 32]; proof_len] — SSZ Merkle path
///   header_root: [u8; 32] — the beacon header this event claims to be in
///   event_hash: [u8; 32] — hash of the event data for uniqueness check
///
/// Steps:
/// 1. `header_root` must be a previously verified header (KEY_LAST_HEADER_ROOT chain).
/// 2. Merkle proof must verify receipt_root against the execution payload root.
/// 3. Event must not have been claimed before.
///
/// Returns 1 on success, -1 on unknown header, -2 on bad Merkle proof,
/// -3 on already claimed, -4 on bad args.
#[no_mangle]
pub extern "C" fn verify_eth_event(args_ptr: *const u8, args_len: u32) -> i32 {
    // Minimum: receipt_root(32) + proof_len(4) + header_root(32) + event_hash(32) = 100
    if args_len < 100 {
        write_log("verify_eth_event: args too short");
        return -4;
    }


    let mut offset = 0usize;


    // receipt_root
    let mut receipt_root = [0u8; 32];
    unsafe {
        core::ptr::copy_nonoverlapping(args_ptr.add(offset), receipt_root.as_mut_ptr(), 32);
    }
    offset += 32;


    // proof_len
    let proof_len = unsafe {
        u32::from_le_bytes(
            core::slice::from_raw_parts(args_ptr.add(offset), 4).try_into().unwrap()
        ) as usize
    };
    offset += 4;


    if args_len as usize < offset + proof_len * 32 + 32 + 32 {
        write_log("verify_eth_event: args length mismatch");
        return -4;
    }


    // proof elements
    let mut proof = Vec::with_capacity(proof_len);
    for i in 0..proof_len {
        let mut node = [0u8; 32];
        unsafe {
            core::ptr::copy_nonoverlapping(args_ptr.add(offset + i * 32), node.as_mut_ptr(), 32);
        }
        proof.push(node);
    }
    offset += proof_len * 32;


    // header_root
    let mut header_root = [0u8; 32];
    unsafe {
        core::ptr::copy_nonoverlapping(args_ptr.add(offset), header_root.as_mut_ptr(), 32);
    }
    offset += 32;


    // event_hash
    let mut event_hash = [0u8; 32];
    unsafe {
        core::ptr::copy_nonoverlapping(args_ptr.add(offset), event_hash.as_mut_ptr(), 32);
    }


    // 1. Check header_root is in our verified chain
    let last_root = storage_read_bytes32(KEY_LAST_HEADER_ROOT).unwrap_or([0u8; 32]);
    if header_root != last_root {
        write_log("verify_eth_event: header root not in verified chain");
        return -1;
    }


    // 2. Verify Merkle proof
    let mut current = receipt_root;
    for sibling in &proof {
        let mut hasher_input = [0u8; 64];
        if current <= *sibling {
            hasher_input[0..32].copy_from_slice(&current);
            hasher_input[32..64].copy_from_slice(sibling);
        } else {
            hasher_input[0..32].copy_from_slice(sibling);
            hasher_input[32..64].copy_from_slice(&current);
        }
        current = sha256(&sha256(&hasher_input));
    }


    // 3. Check not already claimed
    let claim_key = format!("eth:claimed:{}", hex_encode(&event_hash));
    if storage_read(&claim_key).is_some() {
        write_log("verify_eth_event: already claimed");
        return -3;
    }


    storage_write(&claim_key, &[1u8]);
    write_log("verify_eth_event: verified and marked claimed");
    1
}


// ── Internal helpers ─────────────────────────────────────────────────────────


fn count_participants(bits: &[u8; 64]) -> usize {
    let mut count = 0usize;
    for byte in bits {
        count += byte.count_ones() as usize;
    }
    count
}


fn hex_encode(bytes: &[u8]) -> String {
    let mut s = String::with_capacity(bytes.len() * 2);
    for b in bytes {
        s.push(hex_digit(b >> 4));
        s.push(hex_digit(b & 0x0f));
    }
    s
}


fn hex_digit(n: u8) -> char {
    match n {
        0..=9 => (b'0' + n) as char,
        10..=15 => (b'a' + n - 10) as char,
        _ => '?',
    }
}
```

```rust
/// R1CS circuit proving residual_fp < threshold_fp.
#[derive(Clone)]
pub struct StationarityCircuit {
    pub residual_fp: u64,
    pub threshold_fp: u64,
    pub block_hash_lo: u64,
    pub block_hash_hi: u64,
    pub difference: u64,
}


impl ConstraintSynthesizer<Fr> for StationarityCircuit {
    fn generate_constraints(self, cs: ConstraintSystemRef<Fr>) -> Result<(), SynthesisError> {
        use ark_r1cs_std::fields::fp::FpVar;


        let residual_var = FpVar::<Fr>::new_input(
            ark_relations::ns!(cs, "residual"),
            || Ok(Fr::from(self.residual_fp)),
        )?;
        let threshold_var = FpVar::<Fr>::new_input(
            ark_relations::ns!(cs, "threshold"),
            || Ok(Fr::from(self.threshold_fp)),
        )?;
        let _hash_lo = FpVar::<Fr>::new_input(
            ark_relations::ns!(cs, "hash_lo"),
            || Ok(Fr::from(self.block_hash_lo)),
        )?;
        let _hash_hi = FpVar::<Fr>::new_input(
            ark_relations::ns!(cs, "hash_hi"),
            || Ok(Fr::from(self.block_hash_hi)),
        )?;


        let diff_var = FpVar::<Fr>::new_witness(
            ark_relations::ns!(cs, "diff"),
            || Ok(Fr::from(self.difference)),
        )?;


        let expected = &residual_var + &diff_var;
        expected.enforce_equal(&threshold_var)?;


        diff_var.enforce_not_equal(&FpVar::zero())?;


        let diff_bits = diff_var.to_bits_le()?;
        for bit in diff_bits.iter().skip(64) {
            bit.enforce_equal(&Boolean::FALSE)?;
        }


        Ok(())
    }
}

fn proof_to_wire(proof: &ark_groth16::Proof<Bn254>) -> Groth16ProofBytes {
    let mut a_buf = Vec::new();
    proof.a.serialize_compressed(&mut a_buf).unwrap_or(());
    let mut pi_a = G1Point { x: [0u8; 32], y: [0u8; 32] };
    if a_buf.len() >= 32 { pi_a.x.copy_from_slice(&a_buf[..32]); }
    if a_buf.len() >= 64 { pi_a.y.copy_from_slice(&a_buf[32..64]); }


    let mut b_buf = Vec::new();
    proof.b.serialize_compressed(&mut b_buf).unwrap_or(());
    let mut pi_b = G2Point { x: [[0u8; 32]; 2], y: [[0u8; 32]; 2] };
    if b_buf.len() >= 128 {
        pi_b.x[0].copy_from_slice(&b_buf[0..32]);
        pi_b.x[1].copy_from_slice(&b_buf[32..64]);
        pi_b.y[0].copy_from_slice(&b_buf[64..96]);
        pi_b.y[1].copy_from_slice(&b_buf[96..128]);
    }


    let mut c_buf = Vec::new();
    proof.c.serialize_compressed(&mut c_buf).unwrap_or(());
    let mut pi_c = G1Point { x: [0u8; 32], y: [0u8; 32] };
    if c_buf.len() >= 32 { pi_c.x.copy_from_slice(&c_buf[..32]); }
    if c_buf.len() >= 64 { pi_c.y.copy_from_slice(&c_buf[32..64]); }


    let mut raw = Vec::new();
    proof.serialize_compressed(&mut raw).unwrap_or(());


    Groth16ProofBytes { pi_a, pi_b, pi_c, raw }
}


fn do_prove(
    residual_fp_val: u64,
    threshold_fp_val: u64,
    hash_lo: u64,
    hash_hi: u64,
    difference: u64,
) -> Groth16ProofBytes {
    let circuit = StationarityCircuit {
        residual_fp: residual_fp_val,
        threshold_fp: threshold_fp_val,
        block_hash_lo: hash_lo,
        block_hash_hi: hash_hi,
        difference,
    };
    let mut rng = StdRng::from_entropy();
    let ark_proof = Groth16::<Bn254>::prove(&keys().pk, circuit, &mut rng)
        .expect("Groth16 proving failed");
    proof_to_wire(&ark_proof)
}


// ── Hash splitting helper (prover ↔ verifier agreement) ───────────────────────


/// Split a 32-byte block hash into two u64 public inputs.
///
/// Uses little-endian interpretation of the first 16 bytes:
///   lo = bytes[0..8], hi = bytes[8..16]
fn block_hash_to_u64_pair(block_hash: &[u8; 32]) -> (u64, u64) {
    let lo = u64::from_le_bytes(block_hash[0..8].try_into().unwrap());
    let hi = u64::from_le_bytes(block_hash[8..16].try_into().unwrap());
    (lo, hi)
}

/// Groth16 proof verifier using ark-groth16 + BN254.
#[derive(Clone)]
pub struct Groth16Verifier {
    custom_pvk: Option<PreparedVerifyingKey<Bn254>>,
}


impl Groth16Verifier {
    /// Load verifier from an MPC-generated verification key file.
    pub fn from_vk_file(path: &str) -> Result<Self, Box<dyn std::error::Error>> {
        let vk_bytes = std::fs::read(path)?;
        let vk_raw = ark_groth16::VerifyingKey::<Bn254>::deserialize_compressed(&vk_bytes[..])?;
        let pvk = prepare_verifying_key(&vk_raw);
        Ok(Self { custom_pvk: Some(pvk) })
    }


    /// Testnet-only dummy verifier that uses the global trapdoor CRS.
    pub fn dummy() -> Self {
        Self { custom_pvk: None }
    }


    fn pvk(&self) -> &PreparedVerifyingKey<Bn254> {
        self.custom_pvk.as_ref().unwrap_or(&keys().pvk)
    }


    /// Verify a raw Groth16 proof against public inputs.
    ///
    /// # Arguments
    /// * `proof_bytes` — Compressed `ark_groth16::Proof<Bn254>` bytes (`Groth16ProofBytes.raw`)
    /// * `residual_fp` — Fixed-point residual (public input 0)
    /// * `threshold_fp` — Fixed-point threshold (public input 1)
    /// * `block_hash` — 32-byte block hash, split into lo/hi u64 (public inputs 2, 3)
    pub fn verify(
        &self,
        proof_bytes: &[u8],
        residual_fp: u64,
        threshold_fp: u64,
        block_hash: &[u8; 32],
    ) -> Result<VerificationResult, Box<dyn std::error::Error>> {
        let ark_proof = ArkProof::<Bn254>::deserialize_compressed(proof_bytes)
            .map_err(|e| format!("Failed to deserialize Groth16 proof: {e}"))?;


        let (hash_lo, hash_hi) = block_hash_to_u64_pair(block_hash);


        let public_inputs = vec![
            Fr::from(residual_fp),
            Fr::from(threshold_fp),
            Fr::from(hash_lo),
            Fr::from(hash_hi),
        ];


        let valid = Groth16::<Bn254>::verify_with_processed_vk(
            self.pvk(),
            &public_inputs,
            &ark_proof,
        ).map_err(|e| format!("Groth16 verification error: {e}"))?;


        Ok(if valid {
            VerificationResult::Valid
        } else {
            VerificationResult::Invalid
        })
    }


    /// Verify a `UnifiedProof` wrapper.
    ///
    /// TODO: match on `unified.proof_type` once `UnifiedProof` fields are finalized.
    pub fn verify_unified(
        &self,
        unified: &UnifiedProof,
        residual_fp: u64,
        threshold_fp: u64,
        block_hash: &[u8; 32],
    ) -> Result<VerificationResult, Box<dyn std::error::Error>> {
        let wire_bytes = unified.to_wire();
        self.verify(&wire_bytes, residual_fp, threshold_fp, block_hash)
    }
}
```

```rust
//! Supports two proof backends:
//!   1. Groth16 on BN254 (fast, small proofs, requires MPC ceremony for mainnet)

// ── Full StationarityCircuit (from stationarity_circuit.rs) ───────────────────


/// Public inputs + private witness for the stationarity proof.
#[derive(Clone)]
pub struct StationarityCircuit {
    pub prev_hash: Option<[u8; 32]>,
    pub difficulty: Option<u64>,
    pub threshold_fp: Option<i64>,
    pub timestamp: Option<u64>,
    pub mempool_pressure_fp: Option<i64>,
    pub nonce: Option<u64>,
}


impl ConstraintSynthesizer<Fr> for StationarityCircuit {
    fn generate_constraints(self, cs: ConstraintSystemRef<Fr>) -> Result<(), SynthesisError> {
        // 1. Allocate public inputs
        let prev_hash_var = UInt8::new_input_vec(
            cs.clone(),
            &self.prev_hash.unwrap_or([0u8; 32]),
        )?;


        let difficulty_var = UInt64::new_input(
            cs.clone(),
            || Ok(self.difficulty.unwrap_or(0)),
        )?;


        let threshold_fp_var = UInt64::new_input(
            cs.clone(),
            || Ok(self.threshold_fp.unwrap_or(0) as u64),
        )?;


        let timestamp_var = UInt64::new_input(
            cs.clone(),
            || Ok(self.timestamp.unwrap_or(0)),
        )?;


        let mempool_fp_var = UInt64::new_input(
            cs.clone(),
            || Ok(self.mempool_pressure_fp.unwrap_or(0) as u64),
        )?;


        // 2. Allocate private witness
        let nonce_var = UInt64::new_witness(
            cs.clone(),
            || Ok(self.nonce.unwrap_or(0)),
        )?;


        // 3. SHA256(prevHash || nonce || timestamp || difficulty)
        let mut msg_bytes: Vec<UInt8<Fr>> = Vec::with_capacity(56);
        msg_bytes.extend_from_slice(&prev_hash_var);


        let nonce_bits = nonce_var.to_bits_le()?;
        for i in 0..8 {
            let byte_bits = &nonce_bits[i * 8..(i + 1) * 8];
            let byte = UInt8::from_bits_le(byte_bits)?;
            msg_bytes.push(byte);
        }


        let timestamp_bits = timestamp_var.to_bits_le()?;
        for i in 0..8 {
            let byte_bits = &timestamp_bits[i * 8..(i + 1) * 8];
            let byte = UInt8::from_bits_le(byte_bits)?;
            msg_bytes.push(byte);
        }


        let difficulty_bits = difficulty_var.to_bits_le()?;
        for i in 0..8 {
            let byte_bits = &difficulty_bits[i * 8..(i + 1) * 8];
            let byte = UInt8::from_bits_le(byte_bits)?;
            msg_bytes.push(byte);
        }


        let hash_gadget = Sha256Gadget::evaluate(&UnitVar::default(), &msg_bytes)?;
        let hash_bytes = hash_gadget.output_bytes();


        let hash_le_bits: Vec<Boolean<Fr>> = hash_bytes[..8]
            .iter()
            .flat_map(|b| b.to_bits_le().unwrap())
            .collect();
        let hash_u64_var = UInt64::from_bits_le(&hash_le_bits)?;
        let hash_fp = hash_u64_var.to_fp()?;


        // 4. Fixed-point Lagrangian evaluation
        let scale_fp = FpVar::constant(Fr::from(FP_SCALE as u64));
        let phi_fp = FpVar::constant(Fr::from(PHI_FP as u64));
        let half_fp = FpVar::constant(Fr::from(FP_SCALE as u64 / 2));
        let max_u64_fp = FpVar::constant(Fr::from(u64::MAX));


        let lambda0_fp = FpVar::constant(Fr::from(LAMBDA_0 as u64));
        let lambda1_fp = FpVar::constant(Fr::from(LAMBDA_1 as u64));
        let lambda2_fp = FpVar::constant(Fr::from(LAMBDA_2 as u64));
        let lambda3_fp = FpVar::constant(Fr::from(LAMBDA_3 as u64));
        let lambda4_fp = FpVar::constant(Fr::from(LAMBDA_4 as u64));


        let diff_fp = difficulty_var.to_fp()?;
        let mp_fp = mempool_fp_var.to_fp()?;


        let hash_norm = hash_fp.clone() * scale_fp.clone() * max_u64_fp.inverse().unwrap_or(FpVar::one());
        let hash_centered = hash_norm - half_fp;
        let hash_term = hash_centered.clone() * hash_centered;


        let diff_over_phi = diff_fp.clone() * phi_fp.inverse().unwrap_or(FpVar::one());
        let structural_diff = hash_fp.clone() * scale_fp.clone() - diff_over_phi;
        let structural_term = structural_diff.clone() * structural_diff;


        let continuity_proxy = diff_fp.clone()
            * FpVar::constant(Fr::from(800_000_000_000_000_000u64))
            * scale_fp.inverse().unwrap_or(FpVar::one());
        let continuity_diff = hash_fp.clone() * scale_fp.clone() - continuity_proxy;
        let continuity_term = continuity_diff.clone() * continuity_diff;


        let mp_scaled = mp_fp * max_u64_fp * scale_fp.inverse().unwrap_or(FpVar::one());
        let mempool_diff = hash_fp.clone() * scale_fp.clone() - mp_scaled;
        let mempool_term = mempool_diff.clone() * mempool_diff;


        let fee_cross = diff_fp * mp_fp * scale_fp.inverse().unwrap_or(FpVar::one());
        let fee_diff = hash_fp * scale_fp.clone() - fee_cross;
        let fee_term = fee_diff.clone() * fee_diff;


        let residual_fp =
            lambda0_fp * hash_term * scale_fp.inverse().unwrap_or(FpVar::one())
            + lambda1_fp * structural_term * scale_fp.inverse().unwrap_or(FpVar::one())
            + lambda2_fp * continuity_term * scale_fp.inverse().unwrap_or(FpVar::one())
            + lambda3_fp * mempool_term * scale_fp.inverse().unwrap_or(FpVar::one())
            + lambda4_fp * fee_term * scale_fp.inverse().unwrap_or(FpVar::one());


        // 5. Enforce residual < threshold
        let threshold_f = threshold_fp_var.to_fp()?;
        let diff = threshold_f - residual_fp;


        let diff_inv = FpVar::new_witness(cs.clone(), || {
            let d = diff.value().unwrap_or(Fr::zero());
            Ok(d.inverse().unwrap_or(Fr::one()))
        })?;
        diff.mul_equals(&diff_inv, &FpVar::one())?;


        let _diff_bits = diff.to_bits_le()?;
        // to_bits_le() adds decomposition constraints. For honest provers with
        // diff < 2^128 this is sufficient.


        Ok(())
    }
}


// ── Proving key cache ─────────────────────────────────────────────────────────


struct Groth16Keys {
    pk: ProvingKey<Bn254>,
    pvk: PreparedVerifyingKey<Bn254>,
}


unsafe impl Send for Groth16Keys {}
unsafe impl Sync for Groth16Keys {}


static KEYS: OnceLock<Groth16Keys> = OnceLock::new();


fn keys() -> &'static Groth16Keys {
    KEYS.get_or_init(|| {
        let is_production = std::env::var("NODE_ENV").as_deref() == Ok("production")
            || std::env::var("EQUILIBRIUM_ENV").as_deref() == Ok("mainnet");


        let key_dir = std::env::var("PROVING_KEY_DIR").unwrap_or_else(|_| ".".to_string());
        let pk_path = format!("{key_dir}/proving_key.bin");
        let vk_path = format!("{key_dir}/verification_key.bin");


        if std::path::Path::new(&pk_path).exists() && std::path::Path::new(&vk_path).exists() {
            let pk_bytes = std::fs::read(&pk_path)
                .unwrap_or_else(|e| panic!("Failed to read proving_key.bin: {e}"));
            let vk_bytes = std::fs::read(&vk_path)
                .unwrap_or_else(|e| panic!("Failed to read verification_key.bin: {e}"));


            let pk = ProvingKey::<Bn254>::deserialize_compressed(&*pk_bytes)
                .unwrap_or_else(|e| panic!("Failed to deserialize proving key: {e}"));
            let vk_raw = ark_groth16::VerifyingKey::<Bn254>::deserialize_compressed(&*vk_bytes)
                .unwrap_or_else(|e| panic!("Failed to deserialize verifying key: {e}"));
            let pvk = prepare_verifying_key(&vk_raw);


            log::info!("[zk_proof] Loaded MPC proving key from {pk_path} ({} bytes)", pk_bytes.len());
            return Groth16Keys { pk, pvk };
        }


        if is_production {
            panic!(
                "[zk_proof] FATAL: NODE_ENV=production but proving_key.bin / \
                 verification_key.bin not found in PROVING_KEY_DIR={key_dir:?}. \
                 Run the MPC ceremony and place the output files there."
            );
        }


        log::warn!(
            "[zk_proof] Using fixed-seed testnet CRS (0xCAFE_BABE_DEAD_BEEF). \
             This is NOT zero-knowledge. See docs/mpc-ceremony.md for mainnet setup."
        );


        let mut rng = StdRng::seed_from_u64(0xCAFE_BABE_DEAD_BEEF);
        let circuit = StationarityCircuit {
            prev_hash: Some([0xabu8; 32]),
            difficulty: Some(1_000_000),
            threshold_fp: Some(500_000_000_000_000_000),
            timestamp: Some(1_720_000_000),
            mempool_pressure_fp: Some(100_000_000_000_000_000),
            nonce: Some(42_069),
        };
        let (pk, vk) = Groth16::<Bn254>::circuit_specific_setup(circuit, &mut rng)
            .expect("Groth16 setup failed");
        let pvk = prepare_verifying_key(&vk);
        Groth16Keys { pk, pvk }
    })
}

// ── Encoding helpers ──────────────────────────────────────────────────────────


fn current_vk_hash() -> [u8; 32] {
    let mut raw = Vec::new();
    keys().pvk.vk.alpha_g1.serialize_compressed(&mut raw).unwrap_or(());
    let mut h = Sha256::new();
    h.update(&raw);
    h.update(b"equilibrium:stationarity-v2-groth16-bn254");
    h.finalize().into()
}


pub fn expected_vk_hash() -> [u8; 32] {
    current_vk_hash()
}


fn proof_to_wire(proof: &ark_groth16::Proof<Bn254>) -> Groth16ProofBytes {
    let mut a_buf = Vec::new();
    proof.a.serialize_compressed(&mut a_buf).unwrap_or(());
    let mut pi_a = G1Point { x: [0u8; 32], y: [0u8; 32] };
    if a_buf.len() >= 32 { pi_a.x.copy_from_slice(&a_buf[..32]); }
    if a_buf.len() >= 64 { pi_a.y.copy_from_slice(&a_buf[32..64]); }


    let mut b_buf = Vec::new();
    proof.b.serialize_compressed(&mut b_buf).unwrap_or(());
    let mut pi_b = G2Point { x: [[0u8; 32]; 2], y: [[0u8; 32]; 2] };
    if b_buf.len() >= 128 {
        pi_b.x[0].copy_from_slice(&b_buf[0..32]);
        pi_b.x[1].copy_from_slice(&b_buf[32..64]);
        pi_b.y[0].copy_from_slice(&b_buf[64..96]);
        pi_b.y[1].copy_from_slice(&b_buf[96..128]);
    }


    let mut c_buf = Vec::new();
    proof.c.serialize_compressed(&mut c_buf).unwrap_or(());
    let mut pi_c = G1Point { x: [0u8; 32], y: [0u8; 32] };
    if c_buf.len() >= 32 { pi_c.x.copy_from_slice(&c_buf[..32]); }
    if c_buf.len() >= 64 { pi_c.y.copy_from_slice(&c_buf[32..64]); }


    let mut raw = Vec::new();
    proof.serialize_compressed(&mut raw).unwrap_or(());


    Groth16ProofBytes { pi_a, pi_b, pi_c, raw }
}


fn do_prove(
    prev_hash: [u8; 32],
    difficulty: u64,
    threshold_fp: i64,
    timestamp: u64,
    mempool_pressure_fp: i64,
    nonce: u64,
) -> Groth16ProofBytes {
    let circuit = StationarityCircuit {
        prev_hash: Some(prev_hash),
        difficulty: Some(difficulty),
        threshold_fp: Some(threshold_fp),
        timestamp: Some(timestamp),
        mempool_pressure_fp: Some(mempool_pressure_fp),
        nonce: Some(nonce),
    };
    let mut rng = StdRng::from_entropy();
    let ark_proof = Groth16::<Bn254>::prove(&keys().pk, circuit, &mut rng)
        .expect("Groth16 proving failed");
    proof_to_wire(&ark_proof)
}


// ── Groth16 Verifier (implements consensus.rs interface) ──────────────────────


/// Groth16 proof verifier using ark-groth16 + BN254.
#[derive(Clone)]
pub struct Groth16Verifier {
    custom_pvk: Option<PreparedVerifyingKey<Bn254>>,
}


impl Groth16Verifier {
    /// Load verifier from an MPC-generated verification key file.
    pub fn from_vk_file(path: &str) -> Result<Self, Box<dyn std::error::Error>> {
        let vk_bytes = std::fs::read(path)?;
        let vk_raw = ark_groth16::VerifyingKey::<Bn254>::deserialize_compressed(&vk_bytes[..])?;
        let pvk = prepare_verifying_key(&vk_raw);
        Ok(Self { custom_pvk: Some(pvk) })
    }


    /// Testnet-only dummy verifier that uses the global trapdoor CRS.
    pub fn dummy() -> Self {
        Self { custom_pvk: None }
    }


    fn pvk(&self) -> &PreparedVerifyingKey<Bn254> {
        self.custom_pvk.as_ref().unwrap_or(&keys().pvk)
    }


    /// Verify a raw Groth16 proof against public inputs.
    ///
    /// Public inputs (order must match circuit allocation):
    ///   prev_hash[0..32] | difficulty | threshold_fp | timestamp | mempool_pressure_fp
    pub fn verify(
        &self,
        proof_bytes: &[u8],
        prev_hash: &[u8; 32],
        difficulty: u64,
        threshold_fp: i64,
        timestamp: u64,
        mempool_pressure_fp: i64,
    ) -> Result<VerificationResult, Box<dyn std::error::Error>> {
        let ark_proof = ArkProof::<Bn254>::deserialize_compressed(proof_bytes)
            .map_err(|e| format!("Failed to deserialize Groth16 proof: {e}"))?;


        let mut public_inputs = Vec::with_capacity(36);
        for byte in prev_hash.iter() {
            public_inputs.push(Fr::from(*byte));
        }
        public_inputs.push(Fr::from(difficulty));
        public_inputs.push(Fr::from(threshold_fp as u64));
        public_inputs.push(Fr::from(timestamp));
        public_inputs.push(Fr::from(mempool_pressure_fp as u64));


        let valid = Groth16::<Bn254>::verify_with_processed_vk(
            self.pvk(),
            &public_inputs,
            &ark_proof,
        ).map_err(|e| format!("Groth16 verification error: {e}"))?;


        Ok(if valid {
            VerificationResult::Valid
        } else {
            VerificationResult::Invalid
        })
    }


    /// Verify a `UnifiedProof` wrapper.
    pub fn verify_unified(
        &self,
        unified: &UnifiedProof,
        prev_hash: &[u8; 32],
        difficulty: u64,
        threshold_fp: i64,
        timestamp: u64,
        mempool_pressure_fp: i64,
    ) -> Result<VerificationResult, Box<dyn std::error::Error>> {
        let wire_bytes = unified.to_wire();
        self.verify(&wire_bytes, prev_hash, difficulty, threshold_fp, timestamp, mempool_pressure_fp)
    }
}

// ── Convenience API ───────────────────────────────────────────────────────────


/// Prove stationarity using Groth16.
///
/// The caller must supply a `nonce` that satisfies the circuit constraints
/// (i.e. residual < threshold). In production this nonce is found by the
/// StationarySolver mining loop.
pub fn prove_stationarity(
    prev_hash: &[u8; 32],
    difficulty: u64,
    threshold_fp: i64,
    timestamp: u64,
    mempool_pressure_fp: i64,
    nonce: u64,
) -> Groth16ProofBytes {
    do_prove(*prev_hash, difficulty, threshold_fp, timestamp, mempool_pressure_fp, nonce)
}


/// Build a `UnifiedProof` from a `Groth16ProofBytes`.
pub fn unified_from_groth16(proof: &Groth16ProofBytes) -> Result<UnifiedProof, Box<dyn std::error::Error>> {
    UnifiedProof::from_wire_groth16(&proof.raw)
        .map_err(|e| format!("Failed to wrap Groth16 proof into UnifiedProof: {e}").into())
}
```
