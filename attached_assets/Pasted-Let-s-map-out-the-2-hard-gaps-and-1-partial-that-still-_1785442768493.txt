use risc0_zkvm::guest::env;
use serde::{Deserialize, Serialize};

/// Header fields needed to re-evaluate residual at fixed nonce (no search).
#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct StationarityInput {
    pub prev_hash: [u8; 32],
    pub merkle_root: [u8; 32],
    pub timestamp: u64,
    pub nonce: u64,
    pub difficulty: u64,
    pub recursion_depth: u32,
    /// Claimed residual (fixed-point, same scale as chain).
    pub residual_fp: u64,
    pub threshold_fp: u64,
    /// Optional: pack more ChainState fields as you tighten parity.
    pub cumulative_work: u64,
    pub height: u64,
}

#[derive(Serialize, Deserialize, Clone, Debug)]
pub struct StationarityOutput {
    pub residual_fp: u64,
    pub threshold_fp: u64,
    pub block_hash_lo: u64,
    pub block_hash_hi: u64,
}

/// Must match mobile_validator::block_hash / MiningWorker.computeBlockHash
fn block_hash(prev: &[u8; 32], nonce: u64, timestamp: u64, difficulty: u64) -> [u8; 32] {
    use sha2::{Digest, Sha256};
    let mut h = Sha256::new();
    h.update(prev);
    h.update(nonce.to_le_bytes());
    h.update(timestamp.to_le_bytes());
    h.update(difficulty.to_le_bytes());
    h.finalize().into()
}

/// Integer-only residual proxy aligned with fixed-point consensus.
/// Expand until it matches StationarySolver::joint_residual_and_gradient
/// for the fields you put in the journal (document any remaining f64-only terms).
fn residual_at_nonce(inp: &StationarityInput) -> u64 {
    use sha2::{Digest, Sha256};
    let mut h = Sha256::new();
    h.update(inp.prev_hash);
    h.update(inp.merkle_root);
    h.update(inp.timestamp.to_le_bytes());
    h.update(inp.nonce.to_le_bytes());
    h.update(inp.difficulty.to_le_bytes());
    h.update(inp.recursion_depth.to_le_bytes());
    let digest = h.finalize();
    // Fold into u64 residual units (deterministic; replace with full FP formula)
    let mut acc = 0u64;
    for chunk in digest.chunks(8) {
        let mut buf = [0u8; 8];
        buf[..chunk.len()].copy_from_slice(chunk);
        acc ^= u64::from_le_bytes(buf);
    }
    acc
}

risc0_zkvm::guest::entry!(main);

fn main() {
    let input: StationarityInput = env::read();

    let recomputed = residual_at_nonce(&input);
    assert_eq!(
        recomputed, input.residual_fp,
        "claimed residual != recomputed residual"
    );
    assert!(
        input.residual_fp < input.threshold_fp,
        "residual not below threshold"
    );

    let bh = block_hash(
        &input.prev_hash,
        input.nonce,
        input.timestamp,
        input.difficulty,
    );
    let lo = u64::from_le_bytes(bh[0..8].try_into().unwrap());
    let hi = u64::from_le_bytes(bh[8..16].try_into().unwrap());

    env::commit(&StationarityOutput {
        residual_fp: input.residual_fp,
        threshold_fp: input.threshold_fp,
        block_hash_lo: lo,
        block_hash_hi: hi,
    });
}
