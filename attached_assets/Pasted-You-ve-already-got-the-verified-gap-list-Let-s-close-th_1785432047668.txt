Fix 1: Mobile BFT Vote Verification

The stub in `mobile_validator.rs` step 5 needs to:
- Parse Ed25519 vote signatures from peer JSON
- Verify quorum (≥⅔ of active validator stake)
- Reject block if quorum not met under `REQUIRE_BFT_VOTES=true`

```rust
//! Mobile validator — BFT vote verification
//! 
//! Step 5: Collect BFT votes from peer gossip, verify signatures,
//! check quorum against active validator set.

use ed25519_dalek::{Signature, Verifier, VerifyingKey};
use sha2::{Digest, Sha256};

/// BFT vote message format (same as TS consensus):
/// `sha256("equilibrium-bft-v1" || block_hash || height)`
const BFT_VOTE_DOMAIN: &[u8] = b"equilibrium-bft-v1";

#[derive(Debug, Clone)]
pub struct BftVote {
    pub validator_addr: String,
    pub block_hash: [u8; 32],
    pub height: u64,
    pub signature: [u8; 64],
}

impl BftVote {
    pub fn verify(&self, pubkey: &[u8; 32]) -> Result<(), String> {
        let vk = VerifyingKey::from_bytes(pubkey)
            .map_err(|e| format!("invalid pubkey: {}", e))?;
        
        let msg = Self::vote_message(&self.block_hash, self.height);
        let sig = Signature::from_bytes(&self.signature);
        
        vk.verify(&msg, &sig)
            .map_err(|e| format!("signature verify failed: {}", e))
    }
    
    fn vote_message(block_hash: &[u8; 32], height: u64) -> Vec<u8> {
        let mut h = Sha256::new();
        h.update(BFT_VOTE_DOMAIN);
        h.update(block_hash);
        h.update(height.to_le_bytes());
        h.finalize().to_vec()
    }
}

/// Check if a set of votes meets quorum (≥⅔ of total bonded stake).
/// 
/// `validators`: map of addr → (pubkey, bonded_stake)
/// `votes`: verified votes (already passed signature check)
/// `total_stake`: sum of all active validator stakes
pub fn check_bft_quorum(
    validators: &HashMap<String, ([u8; 32], u64)>,
    votes: &[BftVote],
    total_stake: u64,
) -> Result<bool, String> {
    if total_stake == 0 {
        return Err("no active validators".into());
    }
    
    let mut voted_stake: u64 = 0;
    let mut seen = std::collections::HashSet::new();
    
    for vote in votes {
        if seen.contains(&vote.validator_addr) {
            continue; // ignore duplicate votes from same validator
        }
        
        let (pubkey, stake) = validators
            .get(&vote.validator_addr)
            .ok_or_else(|| format!("unknown validator: {}", vote.validator_addr))?;
        
        vote.verify(pubkey)?;
        voted_stake += stake;
        seen.insert(vote.validator_addr.clone());
    }
    
    // Quorum: ≥⅔ total stake (fixed-point: 2 * voted ≥ 3 * total is wrong;
    // correct: voted * 3 ≥ total * 2, or voted ≥ (2 * total) / 3)
    let threshold = (2 * total_stake) / 3;
    Ok(voted_stake >= threshold)
}

/// Parse BFT votes from flat JSON array or nested object.
/// 
/// Input formats supported:
/// - Flat: `[{ "validator": "...", "sig": "hex64", "hash": "hex64", "height": N }, ...]`
/// - Nested: `{ "bftVotes": [...] }`
pub fn parse_bft_votes(json_bytes: &[u8]) -> Result<Vec<BftVote>, String> {
    #[derive(serde::Deserialize)]
    struct FlatVote {
        validator: String,
        sig: String,
        hash: String,
        height: u64,
    }
    
    #[derive(serde::Deserialize)]
    struct Nested {
        #[serde(rename = "bftVotes")]
        bft_votes: Option<Vec<FlatVote>>,
    }
    
    // Try nested first
    if let Ok(nested) = serde_json::from_slice::<Nested>(json_bytes) {
        if let Some(votes) = nested.bft_votes {
            return convert_votes(votes);
        }
    }
    
    // Try flat array
    if let Ok(flat) = serde_json::from_slice::<Vec<FlatVote>>(json_bytes) {
        return convert_votes(flat);
    }
    
    Err("failed to parse BFT votes".into())
}

fn convert_votes(flat: Vec<FlatVote>) -> Result<Vec<BftVote>, String> {
    flat.into_iter()
        .map(|v| {
            let mut sig = [0u8; 64];
            let mut hash = [0u8; 32];
            hex::decode_to_slice(&v.sig, &mut sig)
                .map_err(|e| format!("bad sig hex: {}", e))?;
            hex::decode_to_slice(&v.hash, &mut hash)
                .map_err(|e| format!("bad hash hex: {}", e))?;
            Ok(BftVote {
                validator_addr: v.validator,
                block_hash: hash,
                height: v.height,
                signature: sig,
            })
        })
        .collect()
}
```

Fix 2: UnifiedProof in `consensus.rs`

Wire the dual verifier so `validate_block` accepts both Groth16 and ZkVM proofs:

```rust
//! consensus.rs — UnifiedProof dispatch

use crate::zk_proof::{Groth16Verifier, ZkvmVerifier, UnifiedProof, ProofType};

/// Validate a block with unified proof support.
/// 
/// `proof_bytes`: wire format with 1-byte type tag prefix:
/// - `0x01` → Groth16 (backward compatible)
/// - `0x02` → ZkVM (RISC Zero)
pub fn validate_block_unified(
    header: &BlockHeader,
    proof_bytes: &[u8],
    threshold: u64,
    accepted_types: u8, // bitmask: bit 0 = Groth16, bit 1 = ZkVM
) -> Result<bool, String> {
    if proof_bytes.is_empty() {
        return Err("empty proof".into());
    }
    
    let (proof_type, payload) = match proof_bytes[0] {
        0x01 => (ProofType::Groth16, &proof_bytes[1..]),
        0x02 => (ProofType::Zkvm, &proof_bytes[1..]),
        _ => return Err(format!("unknown proof type: 0x{:02x}", proof_bytes[0])),
    };
    
    // Check governance mask
    let type_bit = match proof_type {
        ProofType::Groth16 => 1u8 << 0,
        ProofType::Zkvm => 1u8 << 1,
    };
    if accepted_types & type_bit == 0 {
        return Err(format!("proof type {:?} not accepted by governance", proof_type));
    }
    
    // Verify residual matches header claim
    let claimed_residual = header.residual_fp;
    
    let valid = match proof_type {
        ProofType::Groth16 => {
            let proof = StationarityProof::from_bytes(payload)
                .map_err(|e| format!("deserialize Groth16: {}", e))?;
            proof.verify(header, threshold)
                .map_err(|e| format!("Groth16 verify: {}", e))?
        }
        ProofType::Zkvm => {
            let verifier = ZkvmVerifier::new()
                .map_err(|e| format!("ZkVM init: {}", e))?;
            verifier.verify_journal(payload, claimed_residual, header.hash())
                .map_err(|e| format!("ZkVM verify: {}", e))?
        }
    };
    
    Ok(valid)
}

/// Backward-compatible wrapper for existing callers.
pub fn validate_block(
    header: &BlockHeader,
    proof: &StationarityProof,
    threshold: u64,
) -> Result<bool, String> {
    let bytes = proof.to_bytes()
        .map_err(|e| format!("serialize: {}", e))?;
    let mut unified = vec![0x01]; // Groth16 tag
    unified.extend_from_slice(&bytes);
    validate_block_unified(header, &unified, threshold, 0xFF) // accept all
}
```

```rust
// Vote message (exact TS layout):
//   Buffer.from("equilibrium-bft-v1")
//   || Buffer.from(block.hash)          // hex STRING, not raw 32 bytes

use std::collections::{HashMap, HashSet};
use ed25519_dalek::{Signature, Verifier, VerifyingKey};

const BFT_VOTE_DOMAIN: &[u8] = b"equilibrium-bft-v1";

#[derive(Debug, Clone)]

    pub validator_addr: String,
    pub block_hash_hex: String, // same string the signer used
    pub height: u64,
    pub signature: [u8; 64],
}

    pub pubkey: [u8; 32],

}

impl BftVote {
    /// Message bytes must match artifacts/api-server/src/chain/state.ts exactly.
    pub fn vote_message(block_hash_hex: &str, height: u64) -> Vec<u8> {
        let mut msg = Vec::with_capacity(BFT_VOTE_DOMAIN.len() + block_hash_hex.len() + 20);
        msg.extend_from_slice(BFT_VOTE_DOMAIN);
        msg.extend_from_slice(block_hash_hex.as_bytes());
        msg.extend_from_slice(height.to_string().as_bytes());
        msg
    }

    pub fn verify(&self, pubkey: &[u8; 32]) -> Result<(), String> {
        let vk = VerifyingKey::from_bytes(pubkey)
            .map_err(|e| format!("invalid pubkey: {e}"))?;
        let msg = Self::vote_message(&self.block_hash_hex, self.height);
        let sig = Signature::from_bytes(&self.signature);
        vk.verify(&msg, &sig)
            .map_err(|e| format!("signature verify failed: {e}"))
    }
}

fn require_bft_votes() -> bool {
    std::env::var("REQUIRE_BFT_VOTES")
        .map(|s| s == "1" || s.eq_ignore_ascii_case("true"))
        .unwrap_or(false)

/// Quorum: voted_stake * 3 >= total_stake * 2  (integer ≥ ⅔)
pub fn check_bft_quorum(
    validators: &HashMap<String, ValidatorInfo>,
    votes: &[BftVote],
    expected_hash_hex: &str,
    expected_height: u64,
) -> Result<(), String> {
    if validators.is_empty() {
        return Err("no active validators in set".into());
    }
    let total_stake: u64 = validators.values().map(|v| v.bonded_stake).sum();
    if total_stake == 0 {
        return Err("total bonded stake is zero".into());
    }

    let mut voted_stake: u64 = 0;
    let mut seen = HashSet::new();

    for vote in votes {
        if !seen.insert(vote.validator_addr.clone()) {
            continue;
        }
        if vote.block_hash_hex != expected_hash_hex || vote.height != expected_height {
            return Err(format!(
                "vote for wrong block: {}@{} vs {}@{}",
                vote.block_hash_hex, vote.height, expected_hash_hex, expected_height
            ));
        }
        let info = validators
            .get(&vote.validator_addr)
            .ok_or_else(|| format!("unknown validator: {}", vote.validator_addr))?;
        vote.verify(&info.pubkey)?;
        voted_stake = voted_stake.saturating_add(info.bonded_stake);
    }

    // voted/total >= 2/3  ⇔  voted*3 >= total*2
    if voted_stake.saturating_mul(3) < total_stake.saturating_mul(2) {
        return Err(format!(
            "BFT quorum not met: voted_stake={voted_stake} total_stake={total_stake}"
        ));
    }
    Ok(())
}

/// Parse FinalityVote-shaped JSON from block body.
/// Accepts top-level `bftVotes` / `bft_votes` array.
pub fn parse_bft_votes_from_block_json(json: &str) -> Vec<BftVote> {
    #[derive(serde::Deserialize)]
    struct WireVote {
        #[serde(alias = "validatorAddress", alias = "validator")]
        validator: String,
        #[serde(alias = "blockHash", alias = "hash")]
        block_hash: String,
        height: u64,
        #[serde(alias = "signature", alias = "sig")]
        signature: String,
    }
    #[derive(serde::Deserialize)]
    struct Envelope {
        #[serde(default, alias = "bftVotes", alias = "bft_votes")]
        bft_votes: Option<Vec<WireVote>>,
    }

    let env: Envelope = match serde_json::from_str(json) {
        Ok(e) => e,
        Err(_) => return vec![],
    };
    let Some(list) = env.bft_votes else { return vec![] };

    let mut out = Vec::with_capacity(list.len());
    for v in list {
        let clean = v.signature.trim().trim_start_matches("0x");
        if clean.len() != 128 || !clean.chars().all(|c| c.is_ascii_hexdigit()) {
            continue;
        }
        let mut sig = [0u8; 64];
        let mut ok = true;
        for i in 0..64 {
            if let Ok(b) = u8::from_str_radix(&clean[i * 2..i * 2 + 2], 16) {
                sig[i] = b;
            } else {
                ok = false;
                break;
            }
        }
        if !ok {
            continue;
        }
        out.push(BftVote {
            validator_addr: v.validator,
            block_hash_hex: v.block_hash.trim().trim_start_matches("0x").to_lowercase(),
            height: v.height,
            signature: sig,
        });
    }
    out
}
```

## Fix 2 — `UnifiedProof` dispatch in `consensus.rs`

Replace / extend `consensus.rs` (keep existing `validate_block` for callers):

```rust
use crate::chain_state::{residual_to_fixed, BlockHeader, ChainState, TxCandidate};
use crate::proof::{accepted_proof_types, ProofType, UnifiedProof, VerificationResult};
use crate::zk_proof::{Groth16Verifier, StationarityProof, ZkvmVerifier};

pub struct Consensus;

impl Consensus {
    /// Legacy path — Groth16 StationarityProof only.
    pub fn validate_block(
        header: &BlockHeader,
        txs: &[TxCandidate],
        proof: &StationarityProof,
        prev_state: &ChainState,
        current_state: &ChainState,
    ) -> bool {
        Self::validate_block_with_mask(header, txs, proof, prev_state, current_state, 0x01)
    }

    pub fn validate_block_with_mask(
        header: &BlockHeader,
        _txs: &[TxCandidate],
        proof: &StationarityProof,
        prev_state: &ChainState,
        current_state: &ChainState,
        accepted_mask: u8,
    ) -> bool {
        if accepted_mask & 0x01 == 0 {
            return false;
        }
        if !StationarityProof::verify(proof, header, current_state, 1e-8) {
            return false;
        }
        if header.residual > residual_to_fixed(1e-8) {
            return false;
        }
        if prev_state.cumulative_work == 0 && header.prev_hash != [0u8; 32] {
            return false;
        }
        true
    }

    /// Dual-proof path. `unified.bytes` is the raw proof payload (no type tag).
    /// `accepted_mask`: bit0 Groth16, bit1 ZkVM (see proof::accepted_proof_types).
    pub fn validate_unified(
        header: &BlockHeader,
        unified: &UnifiedProof,
        prev_state: &ChainState,
        threshold: f64,
        accepted_mask: u8,
    ) -> bool {
        let allowed = accepted_proof_types(accepted_mask);
        if !allowed.contains(&unified.proof_type) {
            return false;
        }

        if header.residual > residual_to_fixed(threshold) {
            return false;
        }
        if prev_state.cumulative_work == 0 && header.prev_hash != [0u8; 32] {
            return false;
        }

        // claimed residual must match header (fixed-point abs for circuit side)
        let residual_fp = header.residual.unsigned_abs();
        if unified.claimed_residual_fp != 0
            && unified.claimed_residual_fp.unsigned_abs() != residual_fp
            && unified.claimed_residual_fp != header.residual
        {
            // allow 0 as "not filled" from from_wire_groth16
            if unified.claimed_residual_fp != 0 {
                return false;
            }
        }

        let threshold_fp = residual_to_fixed(threshold).unsigned_abs();
        // Public inputs bind prev_hash limbs in the circuit
        let block_hash = &header.prev_hash;

        let result = match unified.proof_type {
            ProofType::Groth16 => Groth16Verifier::dummy().verify(
                &unified.bytes,
                residual_fp,
                threshold_fp,
                block_hash,
            ),
            ProofType::Zkvm => ZkvmVerifier::new().verify(
                &unified.bytes,
                residual_fp,
                threshold_fp,
                block_hash,
            ),
        };

        matches!(result, Ok(VerificationResult::Valid))
    }

    pub fn choose_fork(blocks: &[(BlockHeader, i64)]) -> &BlockHeader {
        let idx = blocks
            .iter()
            .enumerate()
            .min_by_key(|(_, (_, r))| *r)
            .map(|(i, _)| i)
            .unwrap_or(0);
        &blocks[idx].0
    }
}
```

The `verify_unified` Bug (Confirmed)

Current code in `zk_proof.rs`:

```rust
pub fn verify_unified(&self, unified: &UnifiedProof, ...) -> Result<VerificationResult, String> {
    let wire_bytes = unified.to_wire();  // prepends 0x01 tag!
    self.verify(&wire_bytes, ...)        // tries to deserialize tag + proof as ark proof
}
```

`to_wire()` produces: `[0x01, <192 bytes proof>, <public inputs>]`

`verify()` feeds all of this into `ArkProof::<Bn254>::deserialize_compressed()`, which fails because the first byte is `0x01` (not a valid compressed proof prefix).

Patch fixes this by using `unified.bytes` directly in `validate_unified`:

```rust
Groth16Verifier::dummy().verify(&unified.bytes, ...)  // no tag
```

```rust
// Vote message (EXACT TS layout — do not use raw hash bytes or LE height):
//   Buffer.from("equilibrium-bft-v1")
//   || Buffer.from(block.hash)           // hex STRING as UTF-8
//   || Buffer.from(block.height.toString())

// Env:

use std::collections::{HashMap, HashSet};
use ed25519_dalek::{Signature, Verifier, VerifyingKey};

const BFT_VOTE_DOMAIN: &[u8] = b"equilibrium-bft-v1";

#[derive(Debug, Clone)]
pub struct ValidatorInfo {
    pub pubkey: [u8; 32],
    pub bonded_stake: u64,
}

#[derive(Debug, Clone)]
pub struct BftVote {
    pub validator_addr: String,
    /// Hex string identical to what the signer hashed (no 0x, lowercase preferred).
    pub block_hash_hex: String,
    pub height: u64,
    pub signature: [u8; 64],
}

impl BftVote {
    pub fn vote_message(block_hash_hex: &str, height: u64) -> Vec<u8> {
        let mut msg = Vec::with_capacity(BFT_VOTE_DOMAIN.len() + block_hash_hex.len() + 24);
        msg.extend_from_slice(BFT_VOTE_DOMAIN);
        msg.extend_from_slice(block_hash_hex.as_bytes());
        msg.extend_from_slice(height.to_string().as_bytes());
        msg
    }

    pub fn verify(&self, pubkey: &[u8; 32]) -> Result<(), String> {
        let vk = VerifyingKey::from_bytes(pubkey)
            .map_err(|e| format!("invalid pubkey: {e}"))?;
        let msg = Self::vote_message(&self.block_hash_hex, self.height);
        let sig = Signature::from_bytes(&self.signature);
        vk.verify(&msg, &sig)
            .map_err(|e| format!("signature verify failed: {e}"))
    }
}

fn require_bft_votes() -> bool {
    std::env::var("REQUIRE_BFT_VOTES")
        .map(|s| s == "1" || s.eq_ignore_ascii_case("true"))
        .unwrap_or(false)
}

/// Quorum: voted_stake * 3 >= total_stake * 2
pub fn check_bft_quorum(
    validators: &HashMap<String, ValidatorInfo>,
    votes: &[BftVote],
    expected_hash_hex: &str,
    expected_height: u64,
) -> Result<(), String> {
    if validators.is_empty() {
        return Err("no active validators in set".into());
    }
    let total_stake: u64 = validators.values().map(|v| v.bonded_stake).sum();
    if total_stake == 0 {
        return Err("total bonded stake is zero".into());
    }

    let mut voted_stake: u64 = 0;
    let mut seen = HashSet::new();

    for vote in votes {
        if !seen.insert(vote.validator_addr.clone()) {
            continue;
        }
        // Normalize hash compare (strip 0x, lowercase)
        let vh = vote
            .block_hash_hex
            .trim()
            .trim_start_matches("0x")
            .trim_start_matches("0X")
            .to_lowercase();
        let eh = expected_hash_hex
            .trim()
            .trim_start_matches("0x")
            .trim_start_matches("0X")
            .to_lowercase();
        if vh != eh || vote.height != expected_height {
            return Err(format!(
                "vote for wrong block: {}@{} vs {}@{}",
                vote.block_hash_hex, vote.height, expected_hash_hex, expected_height
            ));
        }
        let info = validators
            .get(&vote.validator_addr)
            .ok_or_else(|| format!("unknown validator: {}", vote.validator_addr))?;
        vote.verify(&info.pubkey)?;
        voted_stake = voted_stake.saturating_add(info.bonded_stake);
    }

    if voted_stake.saturating_mul(3) < total_stake.saturating_mul(2) {
        return Err(format!(
            "BFT quorum not met: voted_stake={voted_stake} total_stake={total_stake}"
        ));
    }
    Ok(())
}

pub fn parse_bft_votes_from_block_json(json: &str) -> Vec<BftVote> {
    #[derive(Deserialize)]
    struct WireVote {
        #[serde(alias = "validatorAddress", alias = "validator")]
        validator: String,
        #[serde(alias = "blockHash", alias = "hash")]
        block_hash: String,
        height: u64,
        #[serde(alias = "signature", alias = "sig")]
        signature: String,
    }
    #[derive(Deserialize)]
    struct Envelope {
        #[serde(default, alias = "bftVotes", alias = "bft_votes")]
        bft_votes: Option<Vec<WireVote>>,
    }

    let env: Envelope = match serde_json::from_str(json) {
        Ok(e) => e,
        Err(_) => return vec![],
    };
    let Some(list) = env.bft_votes else {
        return vec![];
    };

    let mut out = Vec::with_capacity(list.len());
    for v in list {
        let clean = v
            .signature
            .trim()
            .trim_start_matches("0x")
            .trim_start_matches("0X");
        if clean.len() != 128 || !clean.chars().all(|c| c.is_ascii_hexdigit()) {
            continue;
        }
        let mut sig = [0u8; 64];
        let mut ok = true;
        for i in 0..64 {
            match u8::from_str_radix(&clean[i * 2..i * 2 + 2], 16) {
                Ok(b) => sig[i] = b,
                Err(_) => {
                    ok = false;
                    break;
                }
            }
        }
        if !ok {
            continue;
        }
        out.push(BftVote {
            validator_addr: v.validator,
            block_hash_hex: v
                .block_hash
                .trim()
                .trim_start_matches("0x")
                .trim_start_matches("0X")
                .to_lowercase(),
            height: v.height,
            signature: sig,
        });
    }
    out
}

        .map(|s| s == "1" || s.eq_ignore_ascii_case("true"))
        .unwrap_or(false)
}
```

## Fix 2 — `consensus.rs` (real APIs)

```rust
use crate::chain_state::{residual_to_fixed, BlockHeader, ChainState, TxCandidate};
use crate::proof::{accepted_proof_types, ProofType, UnifiedProof, VerificationResult};
use crate::zk_proof::{Groth16Verifier, StationarityProof, ZkvmVerifier};

pub struct Consensus;

impl Consensus {
    /// Legacy path — Groth16 StationarityProof only (existing callers unchanged).
    pub fn validate_block(
        header: &BlockHeader,
        txs: &[TxCandidate],
        proof: &StationarityProof,
        prev_state: &ChainState,
        current_state: &ChainState,
    ) -> bool {
        Self::validate_block_with_mask(header, txs, proof, prev_state, current_state, 0x01)
    }

    pub fn validate_block_with_mask(
        header: &BlockHeader,
        _txs: &[TxCandidate],
        proof: &StationarityProof,
        prev_state: &ChainState,
        current_state: &ChainState,
        accepted_mask: u8,
    ) -> bool {
        if accepted_mask & 0x01 == 0 {
            return false;
        }
        if !StationarityProof::verify(proof, header, current_state, 1e-8) {
            return false;
        }
        if header.residual > residual_to_fixed(1e-8) {
            return false;
        }
        if prev_state.cumulative_work == 0 && header.prev_hash != [0u8; 32] {
            return false;
        }
        true
    }

    /// Dual-proof path. `unified.bytes` is the raw proof (NO type tag).
    /// `accepted_mask`: bit0 Groth16, bit1 ZkVM.
    pub fn validate_unified(
        header: &BlockHeader,
        unified: &UnifiedProof,
        prev_state: &ChainState,
        threshold: f64,
        accepted_mask: u8,
    ) -> bool {
        let allowed = crate::proof::accepted_proof_types(accepted_mask);
        if !allowed.contains(&unified.proof_type) {
            return false;
        }

        if header.residual > residual_to_fixed(threshold) {
            return false;
        }
        if prev_state.cumulative_work == 0 && header.prev_hash != [0u8; 32] {
            return false;
        }

        // Optional claim check when filled by prover path
        if unified.claimed_residual_fp != 0 && unified.claimed_residual_fp != header.residual {
            return false;
        }

        let residual_fp = header.residual.unsigned_abs();
        let threshold_fp = residual_to_fixed(threshold).unsigned_abs();
        // Circuit binds prev_hash limbs today
        let block_hash = &header.prev_hash;

        let result = match unified.proof_type {
            ProofType::Groth16 => Groth16Verifier::dummy().verify(
                &unified.bytes,
                residual_fp,
                threshold_fp,
                block_hash,
            ),
            ProofType::Zkvm => ZkvmVerifier::new().verify(
                &unified.bytes,
                residual_fp,
                threshold_fp,
                block_hash,
            ),
        };

        matches!(result, Ok(VerificationResult::Valid))
    }

    pub fn choose_fork(blocks: &[(BlockHeader, i64)]) -> &BlockHeader {
        let idx = blocks
            .iter()
            .enumerate()
            .min_by_key(|(_, (_, r))| *r)
            .map(|(i, _)| i)
            .unwrap_or(0);
        &blocks[idx].0
    }
}
```
